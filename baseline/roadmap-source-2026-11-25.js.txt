            function makeTable() {

                var lastUpdate = document.getElementById('lastUpdate');

                lastUpdate.appendChild(document.createTextNode(roadmap.lastUpdate));

                var _tr_ = document.createElement('tr'), _td_ = document.createElement('td'), _ul_ = document.createElement('ul'), _li_ = document.createElement('li'), _a_ = document.createElement('a');

                var roadmapTable = document.getElementById('roadmapTable');



                roadmap.roadmap.forEach(function (element) {

                    var tr = _tr_.cloneNode(false);

                    makeCell(tr, _td_.cloneNode(false), element.date);

                    makeCell(tr, _td_.cloneNode(false), element.affects);

                    makeCell(tr, _td_.cloneNode(false), element.protocol);

                    makeCell(tr, _td_.cloneNode(false), element.action);



                    if (element.testLink == null) {

                        makeCell(tr, _td_.cloneNode(false), element.howToTest);

                    } else

                    {

                        var testLI = _a_.cloneNode(false);

                        testLI.href = element.testLink;

                        testLI.target = "_blank";

                        testLI.appendChild(document.createTextNode(element.howToTest));

                        var td = _td_.cloneNode(false);

                        td.appendChild(testLI);

                        tr.appendChild(td);

                    }





                    if (element.changeLog != null && element.changeLog.length > 0) {

                        var logList = _ul_.cloneNode(false);

                        element.changeLog.forEach(

                                function (change) {

                                    var li = _li_.cloneNode(false);

                                    var changeText = document.createTextNode(change.change);

                                    if (change.link == null) {

                                        li.appendChild(changeText);

                                    } else {

                                        var changeLink = _a_.cloneNode(false);

                                        changeLink.href = change.link;

                                        changeLink.target = "_blank";

                                        changeLink.appendChild(changeText);

                                        li.appendChild(changeLink);

                                    }

                                    logList.appendChild(li);

                                }

                        );

                        var logTd = _td_.cloneNode(false);

                        logTd.appendChild(logList);

                        tr.appendChild(logTd);

                    }



                    roadmapTable.appendChild(tr);

                });

            }
